# Frontend React Patterns

## Read-Only Mode

Use context to conditionally hide write actions:

```typescript
// --- Correct: ReadOnly-aware component ---
import { useReadOnly } from '@/contexts/ReadOnlyContext';

export function VisionCard({ vision }: { vision: Vision }) {
  const { isReadOnly } = useReadOnly();

  return (
    <Card>
      <h3>{vision.title}</h3>
      <p>{vision.description}</p>
      {!isReadOnly && (
        <Button onClick={() => handleEdit(vision.id)}>Edit</Button>
      )}
    </Card>
  );
}
```

## Data Fetching

Use React hooks with proper error/loading states:

```typescript
// --- Correct: Hook-based fetching ---
export function useVisions() {
  const [visions, setVisions] = useState<Vision[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch('/api/roadmap/visions')
      .then(r => r.json())
      .then(data => setVisions(data.data))
      .finally(() => setLoading(false));
  }, []);

  return { visions, loading };
}
```

```typescript
// --- Wrong: Fetch directly in render path ---
export function RiskPanel() {
  // BAD: No loading state, no error handling, fetch in component body
  const [risks, setRisks] = useState([]);
  fetch('/api/risks').then(r => r.json()).then(setRisks);
  return <div>{risks.map(r => <RiskItem key={r.id} risk={r} />)}</div>;
}
```

## Key Conventions
- **Framework:** Next.js 16 + React 19 + TypeScript strict
- **Styling:** TailwindCSS 4
- **Charts:** Recharts
- **Components:** PascalCase, functional only
- **Types:** `src/types/` directory
- **Contexts:** `src/contexts/` (ReadOnlyContext, etc.)
- **No `any` type** — use explicit types or `unknown`
